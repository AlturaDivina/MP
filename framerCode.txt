// src/components/MercadoPagoEmbed.tsx (actualizado con auto-resize y mejoras)
import * as React from "react"
import { addPropertyControls, ControlType } from "framer"

interface MercadoPagoEmbedProps {
    productId: string
    publicKey: string
    successUrl: string
    pendingUrl: string
    failureUrl: string
    hideTitle: boolean
    width: number | string
    height: number | string
    onRedirect?: (event: any) => void
    sessionIdOverride?: string
    displayMode?: "full" | "cartIconOnly" | "paymentFlowOnly" | "sidebarOnly"
    initialStep?: number
    cartIconColor?: string
    autoResize?: boolean
    maxAutoHeight?: number
    minAutoHeight?: number
    /**
     * Controla cómo se maneja la altura:
     * - "auto": el iframe cambia su altura al contenido (postMessage MP_IFRAME_HEIGHT)
     * - "fixed": usa height (prop) o 100dvh y hace scroll interno
     * - "parent": se ajusta al alto del contenedor padre y mantiene scroll interno si overflow
     */
    scrollMode?: "auto" | "fixed" | "parent" | "viewportFit"
}

export function MercadoPagoEmbed(props: MercadoPagoEmbedProps): React.ReactElement {
    const {
        productId,
        publicKey,
        successUrl,
        pendingUrl,
        failureUrl,
        hideTitle,
        width,
        height,
        onRedirect,
        sessionIdOverride,
        displayMode = "full",
        initialStep,
        cartIconColor,
    autoResize = true,
    maxAutoHeight = 3200,
    minAutoHeight = 400,
    scrollMode = "auto",
    } = props

    const iframeRef = React.useRef<HTMLIFrameElement>(null)
    const [dynamicHeight, setDynamicHeight] = React.useState<number>(typeof height === "number" ? height : 800)
    const [parentHeight, setParentHeight] = React.useState<number | null>(null)
    const enableResize = autoResize && scrollMode === "auto"
    const viewportFitRef = React.useRef<() => void>(()=>{})

    // Observa el tamaño del contenedor padre cuando scrollMode = 'parent'
    React.useLayoutEffect(() => {
        if (scrollMode !== "parent") return
        const node = iframeRef.current?.parentElement
        if (!node) return
        const measure = () => {
            const h = node.clientHeight
            if (h && h > 0) setParentHeight(h)
        }
        measure()
        const ro = new ResizeObserver(measure)
        ro.observe(node)
        return () => ro.disconnect()
    }, [scrollMode])

    // Modo viewportFit: calculamos el espacio visible restante en viewport
    React.useLayoutEffect(() => {
        if (scrollMode !== 'viewportFit') return
        const calc = () => {
            if (!iframeRef.current) return
            const rect = iframeRef.current.getBoundingClientRect()
            const avail = window.innerHeight - rect.top - 4 // margen mínimo
            if (avail > 120) {
                iframeRef.current.style.height = avail + 'px'
            }
        }
        viewportFitRef.current = calc
        calc()
        window.addEventListener('resize', calc)
        window.addEventListener('orientationchange', calc)
        window.addEventListener('scroll', calc, true)
        return () => {
            window.removeEventListener('resize', calc)
            window.removeEventListener('orientationchange', calc)
            window.removeEventListener('scroll', calc, true)
        }
    }, [scrollMode])

    // Session global consistente
    const getOrCreateGlobalSessionId = React.useCallback(() => {
        if (typeof window !== "undefined") {
            const urlParams = new URLSearchParams(window.location.search)
            const sessionFromUrl = urlParams.get("sessionId")
            if (sessionFromUrl) return sessionFromUrl
        }
        if (typeof window !== "undefined" && window.sessionStorage) {
            const storedId = sessionStorage.getItem("mp_global_session_id")
            if (storedId) return storedId
        }
        const newSessionId = `session_${Date.now()}_${Math.random().toString(36).substring(2, 11)}`
        if (typeof window !== "undefined" && window.sessionStorage) {
            sessionStorage.setItem("mp_global_session_id", newSessionId)
        }
        return newSessionId
    }, [])

    const sessionId = React.useMemo(
        () => sessionIdOverride || getOrCreateGlobalSessionId(),
        [getOrCreateGlobalSessionId, sessionIdOverride]
    )

    React.useEffect(() => {
        const handleMessage = (event: MessageEvent) => {
            if (!event.origin.includes("mercadopagoiframe.vercel.app")) return
            const { type } = event.data || {}
            if (type === "MP_PAYMENT_SUCCESS" || type === "MP_PAYMENT_PENDING" || type === "MP_PAYMENT_FAILED") {
                const statusMap: Record<string, string> = {
                    MP_PAYMENT_SUCCESS: "success",
                    MP_PAYMENT_PENDING: "pending",
                    MP_PAYMENT_FAILED: "failed",
                }
                const redirectUrl = event.data.redirectUrl
                if (onRedirect) {
                    onRedirect({
                        status: statusMap[type],
                        redirectUrl,
                        data: event.data.paymentData,
                    })
                }
                if (redirectUrl) {
                    window.top.location.href = redirectUrl
                }
            } else if (type === "CART_UPDATE") {
                if (typeof window !== "undefined") {
                    const updateEvent = new CustomEvent("ALTURA_DIVINA_CART_UPDATE", {
                        detail: { source: "mercadopago_iframe", sessionId },
                    })
                    window.dispatchEvent(updateEvent)
                }
            } else if (type === "MP_IFRAME_HEIGHT" && enableResize) {
                const raw = Number(event.data.height)
                if (!isNaN(raw) && raw > 0) {
                    const clamped = Math.min(Math.max(raw, minAutoHeight), maxAutoHeight)
                    setDynamicHeight(clamped)
                    if (iframeRef.current) {
                        iframeRef.current.style.height = `${clamped}px`
                    }
                }
            }
        }
        window.addEventListener("message", handleMessage)
        return () => window.removeEventListener("message", handleMessage)
    }, [onRedirect, sessionId, enableResize, maxAutoHeight, minAutoHeight])

    // Construir src del iframe
    const baseUrl = "https://mercadopagoiframe.vercel.app/"
    const params: Record<string, string> = {
        initialProductId: productId || "",
        publicKey: publicKey || "",
        successUrl: successUrl || "",
        pendingUrl: pendingUrl || "",
        failureUrl: failureUrl || "",
        hideTitle: hideTitle ? "true" : "false",
        isInIframe: "true",
        sessionId: sessionId,
    }
    if (displayMode) params.displayMode = displayMode
    if (initialStep != null) params.initialStep = String(initialStep)
    if (cartIconColor && displayMode === "cartIconOnly") params.cartIconColor = cartIconColor

    const iframeSrc = `${baseUrl}?${new URLSearchParams(params).toString()}`

    const resolvedWidth = typeof width === "number" ? `${width}px` : width || "100%"
    let resolvedHeight: string
    if (scrollMode === "viewportFit") {
        // Altura se setea dinámicamente via efecto; usar fallback inicial
        if (typeof height === 'number') resolvedHeight = `${height}px`
        else if (typeof height === 'string' && height.trim() !== '') resolvedHeight = height
        else resolvedHeight = '100dvh'
    } else if (scrollMode === "parent") {
        if (parentHeight) resolvedHeight = `${parentHeight}px`
        else if (typeof height === "number") resolvedHeight = `${height}px`
        else if (typeof height === "string" && height.trim() !== "") resolvedHeight = height
        else resolvedHeight = "100%"
    } else if (scrollMode === "fixed") {
        if (typeof height === "number") resolvedHeight = `${height}px`
        else if (typeof height === "string" && height.trim() !== "") resolvedHeight = height
        else resolvedHeight = "100dvh"
    } else { // auto
        resolvedHeight = enableResize ? `${dynamicHeight}px` : (typeof height === "number" ? `${height}px` : height || `${dynamicHeight}px`)
    }

    return (
        <iframe
            ref={iframeRef}
            src={iframeSrc}
            width={resolvedWidth}
            height={resolvedHeight}
            style={{
                border: "none",
                borderRadius: "8px",
                overflow: (scrollMode === "fixed" || scrollMode === "parent" || scrollMode === 'viewportFit') ? "auto" : "hidden",
                width: "100%",
                transition: enableResize ? "height 180ms ease" : undefined,
                minHeight: enableResize ? `${minAutoHeight}px` : undefined,
                maxHeight: scrollMode === "parent" && parentHeight ? `${parentHeight}px` : undefined,
            }}
            allow="payment; allow-top-navigation-by-user-activation"
        />
    )
}

// Controles en el panel de Framer
addPropertyControls(MercadoPagoEmbed, {
    productId: { title: "Product ID", type: ControlType.String, defaultValue: "2" },
    publicKey: { title: "MP Public Key", type: ControlType.String, defaultValue: "" },
    successUrl: { title: "Success URL", type: ControlType.String, defaultValue: "https://alturadivina.com/confirmacion-de-compra" },
    pendingUrl: { title: "Pending URL", type: ControlType.String, defaultValue: "https://alturadivina.com/proceso-de-compra" },
    failureUrl: { title: "Failure URL", type: ControlType.String, defaultValue: "https://alturadivina.com/error-de-compra" },
    hideTitle: { title: "Hide Title", type: ControlType.Boolean, defaultValue: false },
    scrollMode: { title: "Scroll Mode", type: ControlType.Enum, options: ["auto", "fixed", "parent", "viewportFit"], optionTitles: ["Auto Resize", "Fixed (inner)", "Lock Parent", "Viewport Fit"], defaultValue: "auto" },
    autoResize: { title: "Auto Resize", type: ControlType.Boolean, defaultValue: true, hidden: (p: any) => p.scrollMode !== 'auto' },
    width: { title: "Width (fallback)", type: ControlType.Number, defaultValue: 400, min: 100, max: 2000, step: 10, hidden: (p: any) => (p.scrollMode === 'auto' && p.autoResize) || p.scrollMode === 'parent' },
    height: { title: "Height (fallback)", type: ControlType.Number, defaultValue: 600, min: 100, max: 3000, step: 10, hidden: (p: any) => (p.scrollMode === 'auto' && p.autoResize) || p.scrollMode === 'parent' },
    minAutoHeight: { title: "Min Auto H", type: ControlType.Number, defaultValue: 400, min: 200, max: 2000, step: 20, hidden: (p: any) => !(p.scrollMode === 'auto' && p.autoResize) },
    maxAutoHeight: { title: "Max Auto H", type: ControlType.Number, defaultValue: 3200, min: 800, max: 6000, step: 50, hidden: (p: any) => !(p.scrollMode === 'auto' && p.autoResize) },
    onRedirect: { title: "On Redirect", type: ControlType.EventHandler },
    sessionIdOverride: { title: "Session ID (opcional)", type: ControlType.String, defaultValue: "" },
    displayMode: { title: "Display Mode", type: ControlType.Enum, options: ["full", "cartIconOnly", "paymentFlowOnly", "sidebarOnly"], defaultValue: "full" },
    cartIconColor: { title: "Cart Icon Color", type: ControlType.Color, defaultValue: "#333333", hidden(props) { return props.displayMode !== "cartIconOnly" } },
    initialStep: { title: "Initial Step (PaymentFlow)", type: ControlType.Number, defaultValue: 1, min: 1, step: 1, hidden(props) { return props.displayMode === "cartIconOnly" } },
})

// Props por defecto en canvas (válidos)
MercadoPagoEmbed.defaultProps = {
    productId: "1",
    publicKey: "",
    successUrl: "https://alturadivina.com/confirmacion-de-compra",
    pendingUrl: "https://alturadivina.com/proceso-de-compra",
    failureUrl: "https://alturadivina.com/error-de-compra",
    hideTitle: false,
    width: 300,
    height: 500,
    displayMode: "full",
    initialStep: 1,
    cartIconColor: "#333333",
    autoResize: true,
    minAutoHeight: 400,
    maxAutoHeight: 3200,
}
